cmake_minimum_required(VERSION 3.20)
project(classroom_translator VERSION 1.0 LANGUAGES CXX)
option(CLASSROOM_NATIVE_OPT "Optimize for this machine" ON)
option(CLASSROOM_ENABLE_TESTS "Build tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(PIPEWIRE REQUIRED IMPORTED_TARGET libpipewire-0.3)
find_package(Threads REQUIRED)
if(NOT EXISTS "${CMAKE_SOURCE_DIR}/third_party/whisper.cpp/CMakeLists.txt")
  message(FATAL_ERROR "Missing third_party/whisper.cpp; run scripts/bootstrap.sh")
endif()
set(GGML_VULKAN ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/whisper.cpp EXCLUDE_FROM_ALL)
add_executable(classroom-translator
 src/main.cpp src/config.cpp src/audio_capture.cpp src/vad_segmenter.cpp
 src/whisper_engine.cpp src/output_sink.cpp src/application.cpp)
target_include_directories(classroom-translator PRIVATE include)
target_link_libraries(classroom-translator PRIVATE PkgConfig::PIPEWIRE whisper Threads::Threads)
target_compile_options(classroom-translator PRIVATE -Wall -Wextra -Wpedantic)
if(CLASSROOM_NATIVE_OPT)
 target_compile_options(classroom-translator PRIVATE $<$<CONFIG:Release>:-O3;-march=native;-mtune=native>)
 target_link_options(classroom-translator PRIVATE $<$<CONFIG:Release>:-flto>)
endif()
if(CLASSROOM_ENABLE_TESTS)
 enable_testing()
 add_executable(test_core tests/test_core.cpp src/vad_segmenter.cpp)
 target_include_directories(test_core PRIVATE include)
 target_link_libraries(test_core PRIVATE Threads::Threads)
 add_test(NAME core COMMAND test_core)
endif()
